\section*{Selected Projects}
\entry{Master's Thesis, OVGU Magdeburg}{}\textit{From RF Embeddings to Semantic Reasoning: Explainable Jamming Analysis via LLM-Based Evidence Interpretation}
\begin{itemize}
\item Raised LLM jamming-regime accuracy from chance (19\%) to \textbf{56\%} (\textbf{73\%} with a zero-cost deterministic correction rule; best of five LLMs 78\%) on a class-balanced 2,500-window test set by replacing salience-ranked embedding evidence with a supervised \textbf{four-axis LDA representation}.
\item Showed that \textbf{100\%-grounded citations were not always causally faithful} by designing paired perturbation tests (masking, necessity, sufficiency, counterfactuals, matched random controls) with McNemar/Holm and group-aware bootstrap statistics across five LLMs.
\item Established the LLM's real contribution by benchmarking it against conventional ML on identical windows: the same four LDA features without an LLM reach 80--81\% (vs.\ 74\% for 768-feature logistic regression), so the LLM adds per-decision explanation, not accuracy.
\end{itemize}
